# jev-duo Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build jev-duo, a two-brain feed agent: an LLM compiles a user's plain-English intent into typed Jev questions, Jev judges every post in a feed in ~100 ms, and the agent folds/dims/badges posts in a Chrome extension (x.com, reddit.com, news.ycombinator.com) and in a CLI over Hacker News.

**Architecture:** One pure TypeScript core (`src/core`) holds the domain model, policy, compiler (System 2), evaluator (System 1), arbiter, learner and provider adapters, all testable without network via mock providers. Two thin surfaces consume the core: a CLI bundled with esbuild, and a Manifest V3 extension whose service worker owns the agent and whose content script owns DOM adapters and the fold UI.

**Tech Stack:** TypeScript 5 (strict, ESM), pnpm, zod, vitest (+ jsdom for DOM tests), esbuild, Playwright (extension + CLI e2e), Chrome Manifest V3, Node >= 20.

**Spec:** `docs/superpowers/specs/2026-09-20-jev-duo-design.md`

## Global Constraints

- Node >= 20, pnpm 10, TypeScript strict, `"type": "module"`; `pnpm typecheck && pnpm test && pnpm build` must pass at the end of every task.
- Fail-open everywhere: any provider or adapter error results in the item being kept visible (`decision.kind === 'keep'`, `source: 'error'`). Never hide content on error.
- v1 hide/keep rules are Noul statements only. The provider layer still supports `choice` and `score` question types.
- Defaults (exact values): hide threshold `0.7`, keep threshold `0.6`, ambiguous band low `0.45`, label max `24` chars, text truncation `2000` chars, LRU cache size `5000`, concurrency `6`, per-item timeout `10000` ms, retries `2` (429 and 5xx only), arbiter budget `20` calls per `600000` ms, example ring buffer `200`, auto-recompile every `10` user examples, strictness clamp `[0.5, 0.95]`.
- Cache key = `sha256(pack.compiledAt + '|' + item.id + '|' + item.text)` hex.
- Jev state is a JSON object `{ platform, author?, text, hasLink, hasMedia, isReply, isPromoted }` (see Task 6 `buildState`).
- Question ids sent to Jev are `r_<ruleId>` for hide rules and `k_<keepId>` for keeps (underscore, never colon: ids are JSON object keys in the TypeSafe request).
- Wire format for Jev is the TypeSafe `POST /v1/systemone` shape documented in `docs/superpowers/research/typesafe-api.md`; the Jev API is called only from Node (CLI) or the extension service worker, never from a content script (the native API rejects browser CORS preflights).
- Installed versions to code against: zod 4.6 (use the string `'custom'` for custom issues), vitest 5, esbuild 0.28, @playwright/test 1.63 (Chromium 1243), TypeScript 5.9.
- CLI exit codes: `0` success, `1` usage error, `2` provider error after retries.
- Extension: Manifest V3, permissions exactly `["storage"]`, no backend, keys only in `chrome.storage.local`.
- All commits end with the trailer line `Co-Authored-By: Claude Fable 5.1 <noreply@anthropic.com>`.
- Implementers never dispatch subagents.
- Write tests first (TDD) for every task that has logic; adapters and UI are tested with jsdom fixtures.

```ts
import { z } from 'zod';
import { DEFAULT_AMBIGUOUS_LOW, DEFAULT_HIDE_THRESHOLD, DEFAULT_KEEP_THRESHOLD, LABEL_MAX_CHARS, STRICTNESS_MAX_THRESHOLD, STRICTNESS_MIN_THRESHOLD } from './constants';
import type { QuestionPack } from './types';

const clamp = (n: number, lo: number, hi: number) => Math.min(hi, Math.max(lo, n));
const Id = z.string().regex(/^[a-z0-9][a-z0-9-]{0,39}$/, 'id must be a lowercase slug');
const Label = z.string().min(1).transform((s) => s.trim().slice(0, LABEL_MAX_CHARS));
const Threshold = z.number().transform((n) => clamp(n, STRICTNESS_MIN_THRESHOLD, STRICTNESS_MAX_THRESHOLD));
const ActionSchema = z.enum(['fold', 'dim', 'badge']);

export const RuleSchema = z.object({
  id: Id, label: z.string().min(1).max(LABEL_MAX_CHARS), question: z.string().min(1),
  threshold: z.number().min(0).max(1), action: ActionSchema,
  ambiguous: z.tuple([z.number().min(0).max(1), z.number().min(0).max(1)]),
});
export const KeepRuleSchema = z.object({ id: Id, label: z.string().min(1).max(LABEL_MAX_CHARS), question: z.string().min(1), threshold: z.number().min(0).max(1) });

const noDuplicateIds = <T extends { id: string }>(xs: T[], ctx: z.RefinementCtx, what: string) => {
  const seen = new Set<string>();
  for (const x of xs) { if (seen.has(x.id)) ctx.addIssue({ code: z.ZodIssueCode.custom, message: `duplicate ${what} id: ${x.id}` }); seen.add(x.id); }
};

export const QuestionPackSchema = z.object({
  version: z.literal(1), intent: z.string(), compiledAt: z.string(), compiledBy: z.string(),
  rules: z.array(RuleSchema), keeps: z.array(KeepRuleSchema), notes: z.string().optional(),
}).superRefine((p, ctx) => { noDuplicateIds(p.rules, ctx, 'rule'); noDuplicateIds(p.keeps, ctx, 'keep'); });

export function parsePack(input: unknown): QuestionPack { return QuestionPackSchema.parse(input) as QuestionPack; }

// What the LLM returns; defaults and normalisation applied here.
export const LlmPackOutputSchema = z.object({
  rules: z.array(z.object({
    id: Id, label: Label, question: z.string().min(1),
    threshold: Threshold.default(DEFAULT_HIDE_THRESHOLD), action: ActionSchema.default('fold'),
  })).transform((rs) => rs.map((r) => ({ ...r, ambiguous: [Math.min(DEFAULT_AMBIGUOUS_LOW, r.threshold), r.threshold] as [number, number] }))),
  keeps: z.array(z.object({ id: Id, label: Label, question: z.string().min(1), threshold: Threshold.default(DEFAULT_KEEP_THRESHOLD) })).default([]),
  notes: z.string().optional(),
});
export type LlmPackOutput = z.infer<typeof LlmPackOutputSchema>;
```

```ts
import { DEFAULT_STRICTNESS, STRICTNESS_MAX_THRESHOLD, STRICTNESS_MIN_THRESHOLD } from './constants';
import type { Decision, QuestionPack, RuleVerdict } from './types';

export interface DecideOptions { strictness?: number } // 0 = lenient, 0.5 = neutral, 1 = strict

const clamp = (n: number, lo: number, hi: number) => Math.min(hi, Math.max(lo, n));

/** Strictness slider maps to a threshold offset of (0.5 - s) * 0.4, clamped to [0.5, 0.95]. */
export function effectiveThreshold(threshold: number, strictness = DEFAULT_STRICTNESS): number {
  return clamp(threshold + (0.5 - strictness) * 0.4, STRICTNESS_MIN_THRESHOLD, STRICTNESS_MAX_THRESHOLD);
}

export function decide(pack: QuestionPack, rules: RuleVerdict[], keeps: RuleVerdict[], opts: DecideOptions = {}): Decision {
  const s = opts.strictness ?? DEFAULT_STRICTNESS;
  const pOf = (xs: RuleVerdict[], id: string) => xs.find((v) => v.ruleId === id)?.p ?? 0;

  for (const k of pack.keeps) if (pOf(keeps, k.id) >= k.threshold) return { kind: 'keep', reason: k.label };

  let best: { p: number; rule: QuestionPack['rules'][number] } | undefined;
  let bestAmbiguous: { p: number; rule: QuestionPack['rules'][number] } | undefined;
  for (const r of pack.rules) {
    const p = pOf(rules, r.id);
    const t = effectiveThreshold(r.threshold, s);
    const low = clamp(r.ambiguous[0] + (0.5 - s) * 0.4, 0.2, t);
    if (p >= t) { if (!best || p > best.p) best = { p, rule: r }; }
    else if (p >= low) { if (!bestAmbiguous || p > bestAmbiguous.p) bestAmbiguous = { p, rule: r }; }
  }
  if (best) return { kind: best.rule.action, ruleId: best.rule.id, label: best.rule.label, p: best.p };
  if (bestAmbiguous) return { kind: 'pending-arbiter', ruleId: bestAmbiguous.rule.id, p: bestAmbiguous.p };
  return { kind: 'keep' };
}
```
